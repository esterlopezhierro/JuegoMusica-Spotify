async function loadPlaylistTracks() { statusLabel.innerText = "Cargando canciones de la playlist..."; let all = []; let url = `https://api.spotify.com/v1/playlists/${playlistId}/items?limit=50`; while (url) { const res = await fetch(url, { headers: { 'Authorization': `Bearer ${accessToken}` } }); if (res.status === 401) { sessionExpired(); return false; } if (!res.ok) { console.error("Error playlist:", res.status, await res.text()); statusLabel.innerText = `No se pudo leer la playlist (error ${res.status}). Debe ser una playlist de tu cuenta.`; return false; } const data = await res.json(); all = all.concat( data.items .map(i => i.item || i.track) .filter(t => t && t.uri && !t.is_local && t.type !== 'episode') ); url = data.next; } playlistTracksCache = all; if (all.length === 0) statusLabel.innerText = "La playlist no tiene canciones."; return all.length > 0; }
